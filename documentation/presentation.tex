% SLIDE 2: MOTIVATION & RESEARCH QUESTION (45s)
% =====================================================================
\begin{frame}{Motivation: Human Learning Is Social}
\begin{columns}[T]
\column{0.55\textwidth}

\textbf{From developmental psychology:}
\begin{itemize}
    \item Caregivers \textbf{curate} the child's information stream (natural pedagogy, Csibra \& Gergely 2009)
    \item They build a \textbf{mental model} of the child (inferential social learning, Gweon 2021)
    \item They \textbf{scaffold} within the Zone of Proximal Development (Vygotsky)
\end{itemize}

\vspace{0.5em}
\textbf{AI agents lack all of this:}
\begin{itemize}
    \item No persistent teacher--learner relationship
    \item No adaptive pedagogical framing
    \item No socially-gated memory consolidation
\end{itemize}

\column{0.42\textwidth}
\begin{block}{Research Question}
Does learning in an \textbf{asymmetric relational dynamic} (caregiver--child) enable an agent to acquire more generalizable knowledge than learning \textbf{alone} or from a \textbf{symmetric peer}?
\end{block}

\vspace{0.5em}
\footnotesize
\textbf{Course connections:} Theory of Mind, cooperative communication, social learning, mental state attribution
\end{columns}

\note{
Human learning is deeply social. From natural pedagogy---Csibra and Gergely's theory that we covered in class---we know caregivers don't just present information randomly; they curate it. They build a mental model of what the child knows, which connects to Theory of Mind and Gweon's inferential social learning framework. And they scaffold within Vygotsky's Zone of Proximal Development.

Current AI agent training has none of this. So I asked: if we give a language agent a persistent caregiver relationship, does it learn better?

This touches on several course topics: Theory of Mind, cooperative communication, social learning, and mental state attribution.
}
\end{frame}

% SLIDE 3: FRAMEWORK (50s)
% =====================================================================
\begin{frame}{Computational Framework}
\begin{columns}[T]
\column{0.50\textwidth}
\textbf{Setup:} 197 household tasks, 8 difficulty levels
\vspace{0.3em}

\textbf{Agents:}
\begin{itemize}
    \footnotesize
    \item \textbf{Caregiver}: Qwen3-235B (frozen, has task solution)
    \item \textbf{Child}: Qwen3-8B + LoRA adapter (learns)
\end{itemize}

\vspace{0.3em}
\textbf{Memory Architecture} (inspired by Tulving 1985):
\begin{enumerate}
    \footnotesize
    \item[\textbf{M1.}] \textbf{Instinct Buffer} -- Fixed role priors
    \item[\textbf{M2.}] \textbf{Working Memory} -- Last 8 turns
    \item[\textbf{M3.}] \textbf{Long-Term Memory} -- Episodic store, BM25 retrieval
    \item[\textbf{M4.}] \textbf{Habits} -- LoRA weights (salience-weighted SFT)
\end{enumerate}

\column{0.46\textwidth}
\textbf{Salience-gated consolidation:}
\vspace{-0.3em}
\[
s = \alpha \!\cdot\! \text{novelty} + \beta \!\cdot\! \text{pred\_err} + \gamma \!\cdot\! \text{teaching}
\]
\vspace{-1em}
\footnotesize
\begin{itemize}
    \item Novelty: information-theoretic surprise
    \item Pred.\ error: Rescorla-Wagner + ZPD match
    \item Teaching: productive struggle signal
    \item \textcolor{red}{$\gamma = 0$ without caregiver}
\end{itemize}

\normalsize
\vspace{0.3em}
\textbf{4 Conditions} (isolate each factor):
\vspace{0.1em}

\footnotesize
\begin{tabular}{lcc}
\toprule
\textbf{Condition} & \textbf{Agents} & $\boldsymbol{\gamma}$ \\
\midrule
Solo & 8B alone & 0 \\
Sym.\ Peer & Two 8B & 0 \\
Role-Labeled & 235B + 8B & 0 \\
\textbf{Relational} & 235B + 8B & \textbf{0.3} \\
\bottomrule
\end{tabular}

\normalsize
\vspace{0.2em}
4 conds $\times$ 3 seeds = 12 concurrent runs
\end{columns}

\note{
Here's the computational framework. I built 197 household tasks across 8 difficulty levels. The caregiver is a frozen 235B model with access to the task solution. The child is an 8B model with a LoRA adapter that gets updated during training.

The memory architecture has four components inspired by Tulving's memory taxonomy from cognitive science: instinct buffer, working memory, long-term episodic memory, and habits implemented as LoRA weights.

The key mechanism is the salience signal---it determines what gets consolidated into long-term memory and how strongly habits form. It combines novelty, prediction error using a Rescorla-Wagner model, and a teaching signal that captures productive struggle. Critically, gamma---the teaching signal weight---is zero without a caregiver. That's the main manipulation.

I compare four conditions to isolate each factor: Solo, Symmetric Peer, Role-Labeled with a caregiver but no teaching salience, and our full Relational condition with teaching salience turned on. Each ran 3 seeds, so 12 total concurrent runs.
}
\end{frame}

\note{
Three hypotheses. H1: the relational condition should produce better transfer to novel tasks. H2: habits should form faster with caregiver guidance. And H3: the child should develop a theory of mind of the caregiver. Let's see what happened.
}

% SLIDE 6: THE KEY FINDING (60s)
% =====================================================================
\begin{frame}{The Key Finding: Scaffolding Dependency}
\begin{columns}[T]
\column{0.42\textwidth}
\includegraphics[width=\textwidth]{figures/h1_transfer.png}

\vspace{0.3em}
\centering
\footnotesize
All conditions: $\sim$0.68--0.71 on held-out tasks\\
\textbf{Differences within noise}

\column{0.55\textwidth}
\vspace{0.5em}
\begin{block}{The Central Finding}
Caregiver agents \textbf{dramatically outperform} during training but show \textbf{no transfer advantage} when tested independently.
\end{block}

\vspace{0.5em}
\textbf{Why?} $\to$ \textbf{Scaffolding Dependency}

\begin{itemize}
    \item LoRA weights encode behaviors for the \emph{assisted} context
    \item Remove the caregiver $\to$ child loses real-time feedback it relied on
    \item Mirrors Wood et al.\ (1976): learners become \textbf{dependent on scaffolding} when it isn't faded
\end{itemize}

\vspace{0.3em}
\footnotesize
\textcolor{gray}{Also: Role-Labeled $\approx$ Relational --- prompt asymmetry captures most of the benefit.}
\end{columns}

\note{
Here's the most interesting finding. When we test the child ALONE on held-out tasks, all four conditions perform the same---around 0.69 completion.

Despite 100 percent training success, the caregiver-trained agents do NOT transfer better. This is scaffolding dependency---a well-documented phenomenon from developmental psychology. Wood, Bruner, and Ross showed learners become dependent on support when it isn't faded. The LoRA weights encode behaviors for the assisted context; remove the caregiver and performance reverts.

Also, Role-Labeled and Relational performed identically---prompt asymmetry alone captures most of the benefit.
}
\end{frame}

\begin{frame}{Takeaway \& Next Steps}

\begin{exampleblock}{Key Takeaway}
Caregiver relationships produce \textbf{dramatically better training} (100\% success, 25\% fewer turns), but agents become \textbf{dependent on scaffolding}---paralleling human developmental psychology.
\end{exampleblock}

\vspace{0.3em}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Course connections:}
\begin{itemize}
    \footnotesize
    \item \textbf{Natural pedagogy} (Csibra \& Gergely) shapes encoding
    \item \textbf{ToM} needed for adaptive teaching
    \item \textbf{Cooperative communication} drives efficiency
    \item Social context changes internal representations
\end{itemize}

\column{0.48\textwidth}
\textbf{Next steps:}
\begin{enumerate}
    \footnotesize
    \item \textbf{Scaffolding fading} -- withdraw caregiver to force independence
    \item \textbf{Semantic ToM eval} -- replace BM25 with model-based similarity
    \item \textbf{Full training runs} -- 41--54 of 160 episodes completed; extend
\end{enumerate}
\end{columns}

\note{
Key takeaway: asymmetric caregiver relationships are powerful for training, but without explicit scaffolding fading, the agent becomes dependent. This parallels human developmental psychology.

Course connections: natural pedagogy explains why caregiver framing changes what gets encoded. Theory of Mind is needed for adaptive teaching. Cooperative communication drives efficiency. And the scaffolding dependency result shows social context fundamentally changes the learner's representations.

Next steps for the final report: implement scaffolding fading, replace BM25 with semantic similarity for ToM, and run full 160-episode training. Thank you!
}
\end{frame}
